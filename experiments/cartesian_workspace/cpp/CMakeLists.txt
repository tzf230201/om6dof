cmake_minimum_required(VERSION 3.16)
project(om6dof_workspace_experiment LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Eigen3 REQUIRED)
find_package(kdl_parser REQUIRED)
find_package(urdf REQUIRED)
add_executable(workspace_scan_cpp scan.cpp)
target_link_libraries(workspace_scan_cpp PRIVATE Eigen3::Eigen kdl_parser::kdl_parser urdf::urdf)
target_compile_options(workspace_scan_cpp PRIVATE -Wall -Wextra -Wpedantic)
add_executable(workspace_report_cpp report.cpp)
target_compile_options(workspace_report_cpp PRIVATE -Wall -Wextra -Wpedantic)
